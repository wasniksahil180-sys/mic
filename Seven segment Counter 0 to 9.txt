#include <reg51.h>

void delay()
{
    int i,j;
    for(i=0;i<100;i++)
        for(j=0;j<1275;j++);
}

void main()
{
    unsigned char a[]={0x3F,0x06,0x5B,0x4F,0x66,
                       0x6D,0x7D,0x07,0x7F,0x6F};
    int i;

    while(1)
    {
        for(i=0;i<10;i++)
        {
            P2=a[i];
            P3=0x40;
            delay();
        }
    }
}
